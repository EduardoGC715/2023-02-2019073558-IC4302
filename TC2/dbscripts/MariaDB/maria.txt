### SCRIPTS ###

## CREATE ##
CREATE DATABASE IF NOT EXISTS pokemon_db;
USE pokemon_db;
CREATE TABLE IF NOT EXISTS pokemon_data (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    type VARCHAR(20) NOT NULL,
    level INT NOT NULL
);
